<script>
"use strict";

const DAYS = [
  "Thứ 2","Thứ 3","Thứ 4","Thứ 5","Thứ 6","Thứ 7","Chủ nhật"
];

const DB_KEY = "smart_schedule_v2";

let db = {users:{},classes:{}};
let currentUser = null;
let registerMode = false;
let teacherDraft = [];
let studentDraft = [];
let editingClassId = null;

function loadDB(){
  try{
    const raw = localStorage.getItem(DB_KEY);
    if(raw){
      db = JSON.parse(raw);
      if(!db.users) db.users = {};
      if(!db.classes) db.classes = {};

      Object.values(db.classes).forEach(cls=>{
        if(!Array.isArray(cls.students)) cls.students = [];
        if(!cls.availability) cls.availability = {};
        if(!Array.isArray(cls.teacherSlots)) cls.teacherSlots = [];
      });
    }
  }catch(e){
    console.error(e);
    db = {users:{},classes:{}};
  }
}

function saveDB(){
  localStorage.setItem(DB_KEY,JSON.stringify(db));
}

function esc(value){
  return String(value ?? "").replace(/[&<>"']/g,c=>({
    "&":"&amp;",
    "<":"&lt;",
    ">":"&gt;",
    '"':"&quot;",
    "'":"&#39;"
  }[c]));
}

function notify(message){
  const el = document.getElementById("toast");
  el.textContent = message;
  el.style.display = "block";
  clearTimeout(notify.timer);
  notify.timer = setTimeout(()=>el.style.display="none",3000);
}

function makeCode(){
  return Math.random().toString(36).slice(2,8).toUpperCase();
}

function teacherClasses(){
  return Object.values(db.classes).filter(c=>c.teacher===currentUser);
}

function joinedClasses(){
  return Object.values(db.classes).filter(c=>
    Array.isArray(c.students) && c.students.includes(currentUser)
  );
}

/* Tạo giao diện lịch rảnh theo từng ngày */
function buildSchedule(containerId,type){
  const container = document.getElementById(containerId);

  container.innerHTML = DAYS.map((day,index)=>`
    <div class="day-card" data-day-card="${index}">
      <div class="day-title">${day}</div>

      <div class="time-inputs">
        <div>
          <label>Từ giờ</label>
          <input type="time" data-start="${index}" value="18:00">
        </div>
        <div>
          <label>Đến giờ</label>
          <input type="time" data-end="${index}" value="20:00">
        </div>
      </div>

      <button type="button" class="add-btn" data-add-day="${index}">
        ＋ Thêm giờ rảnh
      </button>

      <div data-slots="${index}"></div>
    </div>
  `).join("");

  container.querySelectorAll("[data-add-day]").forEach(button=>{
    button.addEventListener("click",()=>{
      addDaySlot(type,Number(button.dataset.addDay),containerId);
    });
  });

  renderSchedule(type,containerId);
}

function draftFor(type){
  return type==="teacher" ? teacherDraft : studentDraft;
}

function addDaySlot(type,day,containerId){
  const container = document.getElementById(containerId);
  const card = container.querySelector(`[data-day-card="${day}"]`);
  const start = card.querySelector(`[data-start="${day}"]`).value;
  const end = card.querySelector(`[data-end="${day}"]`).value;
  const draft = draftFor(type);

  if(!start || !end || start>=end){
    return notify("Giờ kết thúc phải sau giờ bắt đầu.");
  }

  const overlaps = draft.some(s=>
    s.day===day && start<s.end && end>s.start
  );

  if(overlaps){
    return notify("Khoảng giờ này bị trùng với lịch đã thêm.");
  }

  draft.push({day,start,end});
  draft.sort((a,b)=>a.day-b.day || a.start.localeCompare(b.start));
  renderSchedule(type,containerId);
}

function renderSchedule(type,containerId){
  const container = document.getElementById(containerId);
  const draft = draftFor(type);

  DAYS.forEach((day,index)=>{
    const slotContainer = container.querySelector(`[data-slots="${index}"]`);
    const slots = draft.filter(s=>s.day===index);

    if(!slots.length){
      slotContainer.innerHTML = '<p class="small">Chưa thêm giờ rảnh.</p>';
      return;
    }

    slotContainer.innerHTML = slots.map(s=>`
      <div class="slot">
        <span>${esc(s.start)} – ${esc(s.end)}</span>
        <button type="button" data-remove-day="${index}"
          data-remove-start="${esc(s.start)}">×</button>
      </div>
    `).join("");

    slotContainer.querySelectorAll("[data-remove-day]").forEach(button=>{
      button.onclick = ()=>{
        const d = Number(button.dataset.removeDay);
        const start = button.dataset.removeStart;
        const idx = draft.findIndex(s=>s.day===d && s.start===start);
        if(idx>=0) draft.splice(idx,1);
        renderSchedule(type,containerId);
      };
    });
  });
}

document.getElementById("authForm").onsubmit = e=>{
  e.preventDefault();

  const username = document.getElementById("username").value.trim().toLowerCase();
  const password = document.getElementById("password").value;
  const fullName = document.getElementById("fullName").value.trim();

  if(registerMode && !fullName){
    return notify("Vui lòng nhập cả Họ và tên.");
  }

  if(registerMode && fullName.length<2){
    return notify("Họ và tên cần có ít nhất 2 ký tự.");
  }

  if(!/^[a-z0-9_.-]{3,30}$/.test(username)){
    return notify("Tên người dùng cần 3–30 ký tự, gồm chữ thường, số, dấu chấm, _ hoặc -.");
  }

  if(registerMode){
    if(db.users[username]){
      return notify("Tên người dùng đã tồn tại.");
    }

    db.users[username] = {
      password,
      fullName
    };

    saveDB();
  }else{
    if(!db.users[username] || db.users[username].password!==password){
      return notify("Tên người dùng hoặc mật khẩu không đúng.");
    }
  }

  currentUser = username;
  sessionStorage.setItem("smart_schedule_user",username);
  showApp();
};

function showApp(){
  document.getElementById("loginScreen").classList.add("hidden");
  document.getElementById("app").classList.remove("hidden");

  document.getElementById("userLabel").textContent =
    db.users[currentUser]?.fullName || currentUser;

  renderAll();
}

document.getElementById("logout").onclick = ()=>{
  currentUser = null;
  sessionStorage.removeItem("smart_schedule_user");

  document.getElementById("app").classList.add("hidden");
  document.getElementById("loginScreen").classList.remove("hidden");
};

/* Giáo viên tạo lớp */
document.getElementById("createClass").onclick = ()=>{
  const name = document.getElementById("className").value.trim();
  const duration = Number(document.getElementById("lessonDuration").value);

  if(!name) return notify("Hãy nhập tên lớp.");
  if(!teacherDraft.length) return notify("Hãy thêm lịch rảnh của giáo viên.");

  let code = makeCode();
  while(Object.values(db.classes).some(c=>c.code===code)){
    code = makeCode();
  }

  const id = makeCode()+Date.now().toString(36).slice(-5).toUpperCase();

  db.classes[id] = {
    id,
    code,
    name,
    teacher:currentUser,
    duration,
    teacherSlots:teacherDraft.map(s=>({...s})),
    students:[],
    availability:{},
    createdAt:Date.now()
  };

  teacherDraft = [];
  document.getElementById("className").value = "";

  renderSchedule("teacher","teacherSchedule");
  saveDB();
  renderAll();

  notify("Tạo lớp thành công. Mã lớp: "+code);
};

/* Học sinh tham gia lớp */
document.getElementById("joinClass").onclick = ()=>{
  const code = document.getElementById("classCode").value.trim().toUpperCase();
  const cls = Object.values(db.classes).find(c=>c.code===code);

  if(!cls) return notify("Không tìm thấy mã lớp.");
  if(cls.teacher===currentUser){
    return notify("Bạn không thể tham gia lớp do chính mình tạo bằng cùng tài khoản.");
  }

  if(!cls.students.includes(currentUser)){
    cls.students.push(currentUser);
  }

  saveDB();
  renderAll();

  document.getElementById("studentClass").value = cls.id;
  document.getElementById("joinMessage").textContent = "Đã tham gia: "+cls.name;

  studentDraft = cls.availability[currentUser]?.slots
    ? cls.availability[currentUser].slots.map(s=>({...s}))
    : [];

  renderSchedule("student","studentSchedule");
  notify("Tham gia lớp thành công.");
};

/* Khi học sinh đổi lớp, tải lịch đã lưu của lớp đó */
document.getElementById("studentClass").onchange = ()=>{
  const cls = db.classes[document.getElementById("studentClass").value];

  studentDraft = cls?.availability[currentUser]?.slots
    ? cls.availability[currentUser].slots.map(s=>({...s}))
    : [];

  renderSchedule("student","studentSchedule");
};

/* Học sinh gửi lịch rảnh */
document.getElementById("submitSchedule").onclick = ()=>{
  const cls = db.classes[document.getElementById("studentClass").value];

  if(!cls) return notify("Hãy tham gia và chọn một lớp.");
  if(!cls.students.includes(currentUser)){
    return notify("Bạn chưa tham gia lớp này.");
  }
  if(!studentDraft.length){
    return notify("Hãy thêm ít nhất một khung giờ rảnh.");
  }

  cls.availability[currentUser] = {
    slots:studentDraft.map(s=>({...s})),
    updatedAt:Date.now()
  };

  saveDB();
  renderAll();
  notify("Đã gửi lịch rảnh.");
};

/* Các hàm xử lý giờ */
function minutes(time){
  const [h,m] = time.split(":").map(Number);
  return h*60+m;
}

function clock(n){
  return String(Math.floor(n/60)).padStart(2,"0")+":"+
         String(n%60).padStart(2,"0");
}

function mergeRanges(ranges){
  const sorted = ranges.slice().sort((a,b)=>a[0]-b[0]);
  const result = [];

  for(const range of sorted){
    const last = result[result.length-1];

    if(!last || range[0]>last[1]){
      result.push(range.slice());
    }else{
      last[1] = Math.max(last[1],range[1]);
    }
  }

  return result;
}

function intersect(a,b){
  const result = [];

  for(const x of a){
    for(const y of b){
      const start = Math.max(x[0],y[0]);
      const end = Math.min(x[1],y[1]);

      if(end>start) result.push([start,end]);
    }
  }

  return mergeRanges(result);
}

/* Tìm các giờ chung giữa giáo viên và toàn bộ học sinh */
function findBestTimes(cls){
  const students = cls.students;
  const missing = students.filter(u=>!cls.availability[u]?.slots?.length);

  if(!students.length || missing.length){
    return {missing,results:[]};
  }

  const results = [];
  const duration = cls.duration;

  for(let day=0;day<7;day++){
    let common = mergeRanges(
      cls.teacherSlots
        .filter(s=>s.day===day)
        .map(s=>[minutes(s.start),minutes(s.end)])
    );

    for(const username of students){
      const studentRanges = mergeRanges(
        (cls.availability[username]?.slots || [])
          .filter(s=>s.day===day)
          .map(s=>[minutes(s.start),minutes(s.end)])
      );

      common = intersect(common,studentRanges);
      if(!common.length) break;
    }

    for(const [start,end] of common){
      if(end-start<duration) continue;

      for(let t=start;t+duration<=end;t+=15){
        results.push({
          day,
          start:clock(t),
          end:clock(t+duration),
          startMinutes:t,
          duration
        });
      }
    }
  }

  results.sort((a,b)=>a.day-b.day || a.startMinutes-b.startMinutes);
  return {missing,results};
}

function renderTeacherClasses(){
  const classes = teacherClasses();
  const container = document.getElementById("teacherClasses");

  if(!classes.length){
    container.innerHTML = '<p class="muted">Chưa có lớp học.</p>';
    return;
  }

      <strong>${esc(c.name)}</strong>
      <p>Mã lớp: <span class="badge">${esc(c.code)}</span></p>
      <p class="small">Thời lượng: ${c.duration} phút</p>
      <p class="small">
        Học sinh: ${c.students.length}
        · Đã gửi lịch: ${c.students.filter(u=>c.availability[u]?.slots?.length).length}
      </p>
      <button type="button" data-edit-class="${esc(c.id)}">Chỉnh sửa</button>
      <button type="button" data-result="${esc(c.id)}">Xem kết quả</button>
    </div>
  `).join("");

  container.querySelectorAll("[data-edit-class]").forEach(button=>{
    button.onclick = ()=>openClassEditor(button.dataset.editClass);
  });

  container.querySelectorAll("[data-result]").forEach(button=>{
    button.onclick = ()=>{
      document.getElementById("resultClass").value = button.dataset.result;
      renderResults();
      document.getElementById("results").scrollIntoView({behavior:"smooth"});
    };
  });
}

        b.classList.toggle("active",b.dataset.panel==="studentPanel");
      });
      document.getElementById("teacherPanel").classList.add("hidden");
      document.getElementById("studentPanel").classList.remove("hidden");

      notify("Bạn có thể thêm hoặc xóa từng khung giờ. Nhấn Gửi lịch rảnh để lưu.");

    button.onclick = ()=>{
      const cls = db.classes[button.dataset.leaveClass];
      if(!cls) return;

      if(!confirm(
        `Rời khỏi lớp "${cls.name}"? Lịch rảnh của bạn trong lớp này sẽ bị xóa.`

      if(document.getElementById("studentClass").value===cls.id){
        studentDraft = [];
      }

      saveDB();
      renderAll();
      renderSchedule("student","studentSchedule");
      notify("Đã xóa bạn khỏi lớp.");
    };
  });
}

/* Hiển thị kết quả */
function renderResults(){
  const id = document.getElementById("resultClass").value;
  const container = document.getElementById("results");
  const cls = db.classes[id];

  if(!cls || cls.teacher!==currentUser){
    container.innerHTML = '<p class="muted">Chọn lớp để xem kết quả.</p>';
    return;
  }

  const {missing,results} = findBestTimes(cls);
  let html = "";

  if(!cls.students.length){
    html = `<div class="notice">
      Chưa có học sinh tham gia. Hãy gửi mã lớp
      <strong>${esc(cls.code)}</strong> cho học sinh.
    </div>`;
  }else if(missing.length){
    const missingNames = missing.map(u=>db.users[u]?.fullName || u);
    html = `<div class="notice">
      Đang chờ ${missing.length} học sinh gửi lịch:
      ${missingNames.map(esc).join(", ")}.
      Khi tất cả học sinh gửi lịch, hệ thống sẽ tìm giờ chung của cả lớp.
    </div>`;
  }else if(!results.length){
    html = `<div class="notice">
      Không có khoảng thời gian chung đủ ${cls.duration} phút.
      Hãy thử giảm thời lượng buổi học hoặc yêu cầu mọi người bổ sung giờ rảnh.
    </div>`;
  }else{
    html = `
      <div class="result">
        <h3>Các khung giờ phù hợp (${results.length})</h3>
        <p class="muted">
          Các khung giờ được sắp xếp theo thứ tự ngày trong tuần và giờ bắt đầu.
        </p>
        ${results.slice(0,30).map(r=>`
          <div class="result-time">
            <strong>${DAYS[r.day]} · ${r.start} – ${r.end}</strong>
            <p class="small">Thời lượng: ${r.duration} phút</p>
          </div>
        `).join("")}
        ${results.length>30
          ? `<p class="small">Đang hiển thị 30 trên tổng số ${results.length} khung giờ.</p>`
          : ""}
      </div>
    `;
  }

  container.innerHTML = html;
}

/* Chuyển lịch rảnh thành văn bản để chỉnh sửa */
function slotLines(slots){
  return (slots || []).map(s=>
    `${DAYS[s.day]} ${s.start}-${s.end}`
  ).join("\n");
}

/* Đọc và kiểm tra lịch rảnh được nhập trong trình chỉnh sửa */
function parseSlotLines(text){
  const slots = [];
  const dayMap = new Map(DAYS.map((d,i)=>[d.toLowerCase(),i]));

  for(const raw of text.split(/\r?\n/)){
    const line = raw.trim();
    if(!line) continue;

    const match = line.match(
      /^(Thứ\s*[2-7]|Chủ nhật)\s+(\d{2}:\d{2})\s*-\s*(\d{2}:\d{2})$/i
    );

    if(!match){
      throw new Error(`Dòng không hợp lệ: ${line}`);
    }

    const dayText = match[1].replace(/\s+/g," ").toLowerCase();
    const day = dayMap.get(dayText);
    const start = match[2];
    const end = match[3];

    if(!/^\d{2}:\d{2}$/.test(start) || !/^\d{2}:\d{2}$/.test(end)){
      throw new Error(`Định dạng giờ không hợp lệ: ${line}`);
    }

    if(start>="24:00" || end>="24:00"){
      throw new Error(`Giờ phải nằm trong khoảng 00:00–23:59: ${line}`);
    }

    if(start>=end){
      throw new Error(`Giờ kết thúc phải sau giờ bắt đầu: ${line}`);
    }

    if(slots.some(s=>
      s.day===day && start<s.end && end>s.start
    )){
      throw new Error(`Khung giờ bị trùng: ${line}`);
    }

    slots.push({day,start,end});
  }

  if(!slots.length){
    throw new Error("Hãy nhập ít nhất một khung giờ rảnh.");
  }

  return slots.sort((a,b)=>a.day-b.day || a.start.localeCompare(b.start));
}

/* Mở trình chỉnh sửa lớp */
function openClassEditor(id){
  const cls = db.classes[id];
  if(!cls || cls.teacher!==currentUser) return;

  editingClassId = id;

  document.getElementById("editClassName").value = cls.name;
  document.getElementById("editClassDuration").value = String(cls.duration);
  document.getElementById("editTeacherSlots").value = slotLines(cls.teacherSlots);

  const box = document.getElementById("editClassStudents");

  if(!cls.students.length){
    box.innerHTML = '<p class="muted">Chưa có học sinh.</p>';
  }else{
    box.innerHTML = cls.students.map(username=>`
      <div class="slot">
        <span>
          ${esc(db.users[username]?.fullName || username)}
          (${esc(username)})
        </span>
        <button type="button" data-remove-student="${esc(username)}"
          style="background:#b42318">Xóa khỏi lớp</button>
      </div>
    `).join("");
  }

  box.querySelectorAll("[data-remove-student]").forEach(button=>{
    button.onclick = ()=>{
      const username = button.dataset.removeStudent;
      const studentName = db.users[username]?.fullName || username;

      if(!confirm(
        `Xóa ${studentName} khỏi lớp này? Lịch rảnh của học sinh trong lớp sẽ bị xóa.`
      )) return;

      const currentClass = db.classes[editingClassId];
      if(!currentClass) return;

      currentClass.students = currentClass.students.filter(u=>u!==username);
      delete currentClass.availability[username];

      saveDB();
      renderAll();
      openClassEditor(editingClassId);
      notify("Đã xóa học sinh khỏi lớp.");
    };
  });

  document.getElementById("editClassModal").classList.remove("hidden");
}

/* Lưu chỉnh sửa lớp */
document.getElementById("saveClassEdit").onclick = ()=>{
  const cls = db.classes[editingClassId];
  if(!cls || cls.teacher!==currentUser) return;

  const name = document.getElementById("editClassName").value.trim();

  if(!name){
    return notify("Hãy nhập tên lớp.");
  }

  let slots;

  try{
    slots = parseSlotLines(document.getElementById("editTeacherSlots").value);
  }catch(error){
    return notify(error.message);
  }

  cls.name = name;
  cls.duration = Number(document.getElementById("editClassDuration").value);
  cls.teacherSlots = slots;

  saveDB();
  renderAll();

  document.getElementById("editClassModal").classList.add("hidden");
  editingClassId = null;

  notify("Đã cập nhật lớp học.");
};

document.getElementById("deleteAccount").onclick = ()=>{

  )) return;

    if(cls.teacher===username){
      delete db.classes[cls.id];
    }else{

      delete cls.availability[username];

  saveDB();

  document.getElementById("app").classList.add("hidden");
  document.getElementById("loginScreen").classList.remove("hidden");

  document.getElementById("fullName").required = false;

  document.getElementById("authButton").textContent = "Đăng nhập";
  document.getElementById("switchText").textContent = "Chưa có tài khoản?";
  document.getElementById("switchAuth").textContent = "Đăng ký";

  notify("Đã xóa tài khoản.");
};

/* Cập nhật giao diện sau khi thay đổi dữ liệu */
function renderAll(){
  renderTeacherClasses();
  renderJoinedClasses();

  const mine = teacherClasses();
  const select = document.getElementById("resultClass");
  const previous = select.value;

  select.innerHTML = '<option value="">-- Chọn lớp --</option>' +
    mine.map(c=>
      `<option value="${esc(c.id)}">${esc(c.name)} (${esc(c.code)})</option>`
    ).join("");

  if(mine.some(c=>c.id===previous)){
    select.value = previous;
  }else if(mine.length){
    select.value = mine[0].id;
  }

  renderResults();

  const joined = joinedClasses();
  const studentSelect = document.getElementById("studentClass");
  const old = studentSelect.value;

  studentSelect.innerHTML = '<option value="">-- Chọn lớp --</option>' +
    joined.map(c=>
      `<option value="${esc(c.id)}">${esc(c.name)} (${esc(c.code)})</option>`
    ).join("");

  if(joined.some(c=>c.id===old)){
    studentSelect.value = old;
  }else if(joined.length){
    studentSelect.value = joined[0].id;
  }else{
    studentSelect.value = "";
  }
}
